use anchor_lang::prelude::*;

#[program]
pub mod vault {
    use super::*;

    pub fn deposit(ctx: Context<Deposit>, amount: u64) -> Result<()> {
        require!(amount > 0, VaultError::ZeroAmount);
        let vault = &mut ctx.accounts.vault;
        vault.balance = vault.balance
            .checked_add(amount)
            .ok_or(VaultError::Overflow)?;
        Ok(())
    }
}

#[derive(Accounts)]
pub struct Deposit<'info> {
    #[account(mut, seeds = [b"vault", owner.key().as_ref()], bump)]
    pub vault: Account<'info, Vault>,
    pub owner: Signer<'info>,
}

FAQ

Common questions.

Can't find your answer? Email support@necxsoft.com.

Why do you focus on Solana?

Fast confirmations, low fees and one composable state make Solana the best fit for consumer and high-throughput products. Going deep on one ecosystem means sharper expertise in its programs, tooling and security patterns.

Can you work with other chains?

Solana is our focus. If your product also needs an EVM chain, for example to bridge or to support multiple chains, we scope that during discovery.

Do you audit smart contracts?

Every program goes through thorough testing and an internal security review. For mainnet launches that will hold real value, we recommend an independent third-party audit. We prepare your codebase for it and help coordinate the audit.

How long does a project take?

It depends on scope. Mint and landing pages typically take 1–2 weeks, programs and dApps 2–6 weeks, and AI agents and integrations 3–8 weeks.

Do I need to pay before I see anything?

No. We work demo-first, so you see a working devnet preview before any commitment is final.

Do you work with clients outside the Philippines?

Yes. We work with US and international clients entirely online.

What happens after launch?

We offer ongoing support and maintenance, including program upgrades, monitoring and new features as your product grows.

How much does a project cost?

Starter projects begin at $5,000 and Business projects at $15,000. We quote complex protocols individually after discovery.
